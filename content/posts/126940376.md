---这个相对比较简单，就是两个字符串之间的替换

**代码截图：**

![](https://i-blog.csdnimg.cn/blog_migrate/707c1ca5ff38b458272d009922242fe8.png)

 **代码：**

int main() {  
    //定义两个字符串  
    char arr1\[\] = "welcome to small planet";  
    char arr2\[\] = "\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*\*";

    //定义左边第一个下标位置  
    int left = 0;  
    //定义右边最后一个字符下标位置  
    int right = sizeof(arr2)-1;

    //将两边同时替换字符  
    while (left <= right)  
    {  
            //sleep(1000);  
            arr2\[left\] = arr1\[left\];  
            left++;  
            arr2\[right\] = arr1\[right\];  
            right--;

            printf("%s\\n", arr2);  
              
    }  
      
    return 0;  
}
