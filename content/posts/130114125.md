代码：

```cpp
LinkList* reverseList2(LinkList* L) {
    //定义3个指针
    LinkList* pre, *p, *r;
    p = L->next;
    r = p->next;
    while (r) {
        pre = p;//将p指针赋给pre
        p = r;
        r = r->next;
        //将p指针next指向pre，实现反向
        p->next = pre;
    }
    //循环结束后，将头结点L指向p
    L->next = p;
    return L;

}
```
