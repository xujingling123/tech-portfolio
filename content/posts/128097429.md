---比如：i like lay. 倒置后：lay. like i  
思路：1.先将所有字母倒置:.yal ekil i  
2.将1中倒置的再进行倒置（按遇到空格就停下，然后移动到下一位再重新倒置）：lay. like i

![](https://i-blog.csdnimg.cn/blog_migrate/5b58fc5e4383583b5ea53f0e4a796f73.png)

 代码：

void reserse1(char\* start,char\* end) {  
    while (start<end)  
    {  
        char tmp = 0;  
        tmp = \*start;  
        \*start = \*end;  
        \*end = tmp;  
        start++;  
        end--;  
    }  
    

}

//将一个单词倒置，标点符号不倒置  
//比如：i like lay. 倒置后：lay. like i  
//思路：1.先将所有字母倒置:.yal ekil i

void reverse() {  
    char arr\[20\] = {1};  
    //scanf("%s", arr);  
    // 使用scanf遇到空格就不读取打印了  
    //1.先将所有字母倒置:.yal ekil i  
    //使用gets()  
    gets(arr);  
    //求字符串长度  
    int len = strlen(arr);  
    reserse1(arr,arr+len-1);  
    //2.将1中倒置的再进行倒置（按遇到空格就停下，然后移动到下一位再重新倒置）：lay. like i  
    char\* start = arr;  
    while (\*start)  
    {  
        char\* end = start;  
        while (\*end !=' '&&\*end !='\\0')  
        {  
            end++;  
        }  
        //逆序一个单词  
        reserse1(start, end - 1);  
        if (\*end==' ')  
        {  
            start = end + 1;  
        }  
        else  
        {  
            start = end;  
        }  
          
    }

    printf("%s", arr);  
}
