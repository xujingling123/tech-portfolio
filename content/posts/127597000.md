---对应求9\*9乘法口诀表，其实主要是要搞清楚每行和每列

可以用一个for循环来代表每行的第1个因数i：比如

1\*1=1     该行第一个因数为1

2\*1=2     2\*2=4    该行第一个因数为2

3\*1=3     3\*2=6     3\*3=9    该行第一个因数为3

..............

以此类推下去

而我们发现每一列的第2个因数j是相同的，每一行都是依次递增到i的，则可以在原来的for循环里面嵌套一个for循环来求解

![](https://i-blog.csdnimg.cn/blog_migrate/f2f022a2f84e0dcb2b146da9036fd070.png)

 ![](https://i-blog.csdnimg.cn/blog_migrate/008c230b2e4782b0d2f04cab487274fb.png)

函数代码：

//在屏幕上输出9\*9乘法口诀表  
//1\*1=1  
//2\*1=2 2\*2=4  
//3\*1=3 3\*2=6 3\*3=9  
//...  
int mult\_table() {  
    int i = 0;  
    //行  
    for (i = 1; i <= 9; i++) {  
        int j = 1;  
        //列  
        for (j = 1; j <= i; j++) {  
            printf("%d\*%d=%d  ", i, j, i \* j);  
        }  
        printf("\\n");  
    }  
}

//求10 个整数中最大值  
int get\_tenMax(int arr\[\]) {  
    int i = 0;  
    int max = arr\[0\];  
    for (i = 1; i <= 10; i++) {  
        if (arr\[i\] > max) {  
            max = arr\[i\];  
        }  
    }  
    return max;  
}
