//输入任意一个三位数，将其各数字反序输出（例如输入123，输出321）
int main() {
	int n, re=0;
	printf("请输入一个三位数：");
	scanf("%d", &n);
	re = n % 10 * 100 + n / 10 % 10 * 10 + n / 100;
	printf("反序输出结果为：%d", re);
	return 0;
}

//编写一个程序，根据本金a,存款年数n,年利率p计算到期利息
//计算公式：到期利息=a*(1+p)^n-a
#include <math.h>
int main() {
	float a, p,re=0.0;
	int n;
	printf("请输入本金，存款年数，年利率：");
	scanf("%f %d %f", &a, &n, &p);
	re = a * pow(1 + p, n) - a;
	printf("\n到期利息为：%f", re);
	return 0;
}

//编写程序计算圆周长l,圆面积s,圆柱体体积v，并输出计算结果
#define PI 3.1415926
int main() {
	float r, h, l=0, s=0, v=0;
	printf("请输入你要计算的圆柱体半径r，圆柱高h:");
	scanf("%f %f", &r, &h);
	l = 2 * PI * r;
	s = PI * r * r;
	v = s * h;
	printf("圆周长l:%f,圆面积s:%f,圆柱体体积v:%f", l,  s, v);
	return 0;
}
