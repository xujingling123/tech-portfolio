//求两个数的二进制中不同位的个数  
int diff() {  
    int a = 0;  
    int b = 0;  
    int count = 0;  
    int i = 0;  
    scanf("%d %d", &a,&b);  
    for (i = 0; i < 32; i++) {  
        if (((a >> i) & 1) != ((b >> i) & 1)) {  
            count++;  
        }  
    }  
    return count;  
}

int main() {  
    int sum = diff();  
    printf("%d\\n", sum);  
    return 0;  
}
