//求Sn=a+aa+aaa+aaaa+...的前n项之和
#include<stdio.h>
main()
{
	int a, n,i;
	printf("请输入a,n的值：");
	scanf("%d,%d", &a, &n);
	int sum = 0,ret = 0;
	for (i = 0; i < n; i++) {
		ret = ret * 10 + a;//算出每一项
		sum += ret;
	}
	printf("sum=%d\n", sum);


}
