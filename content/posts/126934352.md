---因为用到了输入scanf，记得在代码前面加上**#define \_CRT\_SECURE\_NO\_WARNINGS**，这样运行就不会报错了

**代码**：

#define \_CRT\_SECURE\_NO\_WARNINGS

#include <stdio.h>

int main() {

int i = 0;

int n = 0;

int sum = 1;

scanf("%d", &n);

for (i = 1; i <= n; i++) {

sum \*= i;

}

printf("阶乘是：%d",sum);

return 0;

}
