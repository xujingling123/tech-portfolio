//获取数字a的奇数位和偶数位  
void oddandeven(int a) {  
    //例：7的二进制：00000.....0111  
    //右边第一个数为奇数位，第二个为偶数位，以此类推  
    int i = 0;  
    //偶数位  
    for (i = 31; i >= 1; i-= 2) {  
        printf("%d", (a >> i) & 1);  
    }  
    printf("\\n");  
    //奇数位  
    for (i = 30; i >= 0; i -= 2) {  
        printf("%d", (a >> i) & 1);  
    }

  
}
