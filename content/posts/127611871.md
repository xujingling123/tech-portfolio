#include <stdio.h>

//冒泡排序函数(降序)  
void bubble\_sort(int arr\[\], int sz) {

    int i = 0;  
    for ( i = 0; i < sz-1; i++)  
    {  
        int j = 0;  
        for (j = 0; j < sz - 1 - i; j++) {  
            if (arr\[j\] < arr\[j + 1\]) {  
                int temp = arr\[j\];  
                arr\[j\] = arr\[j + 1\];  
                arr\[j + 1\] = temp;  
            }  
        }

    }

}

  
int main() {

    int arr\[7\] = { 1,2,3,4,5,6,7 };  
    int sz = sizeof(arr) / sizeof(arr\[0\]);  
    bubble\_sort(arr, sz);  
    for (int i = 0; i < sz; i++) {  
        printf("%d", arr\[i\]);

  
    }  
      
    return 0;  
}
