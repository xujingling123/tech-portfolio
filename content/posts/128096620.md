int getdouble1() {  
    int a = 0;  
    int b = 0;  
    scanf("%d %d", &a, &b);

    int min = a > b ? a : b;  
    while (1)  
    {  
        if (min % a == 0 && min % b == 0) {  
            return min;  
        }  
        min++;

    }  
}

方法2（使用乘法的效率要比方法1使用min++的效率高），所以推荐使用方法2：

![](https://i-blog.csdnimg.cn/blog_migrate/30fee5e4996d992f07fbe2717da43b3f.png)

int getdouble2() {  
    int a = 0;  
    int b = 0;  
    int i = 0;  
    scanf("%d %d", &a, &b);

    int min = a > b ? a : b;  
    for ( i = 1;; i++)  
    {  
        if ((min \* i) % a == 0 && (min \* i) % b == 0) {  
            return min \* i;  
        }

    }  
}
